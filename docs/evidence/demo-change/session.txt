gabri@DESKTOP-G4UK84R:~/projects/kiratech-challenge$ (
set -euo pipefail
cd ~/projects/kiratech-challenge
umask 077
minikube start --profile kiratech

out=".local/demo-validation-$(date +%Y%m%d-%H%M%S)"
mkdir -p "$out"

k() { kubectl --context=kiratech -n wordpress-build "$@"; }
cpu() { k get deployment patch-lab-wordpress -o jsonpath='{.spec.template.spec.containers[0].resources.requests.cpu}'; }
image() { k get deployment patch-lab-wordpress -o jsonpath='{.spec.template.spec.containers[0].image}'; }
uploads() { k exec deployment/patch-lab-wordpress -- sh -c 'find /var/www/html/wp-content/uploads -type f -exec sha256sum {} \; | sort'; }
volumes() { k get pvc -o jsonpath='{range .items[*]}{.metadata.name}{" "}{.metadata.uid}{" "}{.spec.volumeName}{" "}{.status.phase}{"\n"}{end}' | sort; }
upgrade() { helm upgrade patch-lab charts/kiratech-wordpress --kube-context kiratech -n wordpress-build --values "$out/values-before.yaml" --set "wordpress.resources.requests.cpu=$1" --wait --timeout 5m; }

helm status patch-lab --kube-context kiratech -n wordpress-build -o json > "$out/helm-before.json"

python3 - "$out/helm-before.json" <<'PY'
import json, sys
with open(sys.argv[1]) as source:
    assert json.load(source)["info"]["status"] == "deployed", "Release is not deployed"
PY

k rollout status deployment/patch-lab-wordpress --timeout=120s
test "$(cpu)" = 100m
test "$(image)" = kiratech-wordpress:7.1.3-php8.3.36-lab

helm get values patch-lab --kube-context kiratech -n wordpress-build --all -o yaml > "$out/values-before.yaml"
uploads > "$out/uploads-before.sha256"
test -s "$out/uploads-before.sha256"
volumes > "$out/volumes-before.txt"
k get deployments,pods,pvc > "$out/resources-before.txt"

restore_needed=0
cleanup() {
  result=$?
  trap - EXIT
  if test "$restore_needed" = 1; then
    if ! upgrade 100m > "$out/emergency-restore.txt" 2>&1; then
      printf 'Restoration failed. Check %s/emergency-restore.txt\n' "$out" >&2
      result=1
    fi
  fi
  exit "$result"
}
trap cleanup EXIT

restore_needed=1
upgrade 150m | tee "$out/upgrade-150m.txt"
test "$(cpu)" = 150m
printf 'CPU request 150m: PASS\n' | tee "$out/cpu-change.txt"

upgrade 100m | tee "$out/restore-100m.txt"
test "$(cpu)" = 100m
restore_needed=0
printf 'CPU request restored to 100m: PASS\n' | tee "$out/cpu-restored.txt"
test "$(image)" = kiratech-wordpress:7.1.3-php8.3.36-lab

uploads > "$out/uploads-after.sha256"
volumes > "$out/volumes-after.txt"
diff -u "$out/uploads-before.sha256" "$out/uploads-after.sha256"
diff -u "$out/volumes-before.txt" "$out/volumes-after.txt"

k exec deployment/patch-lab-wordpress -- \
  curl --fail --silent --show-error --max-time 20 \
  --header 'Host: localhost:8083' \
  --write-out '\nHTTP %{http_code}\n' \
  http://127.0.0.1/ > "$out/page-after.txt"

grep -Fq 'Custom image patch test' "$out/page-after.txt"
test "$(tail -n 1 "$out/page-after.txt")" = 'HTTP 200'

k get deployments,pods,pvc | tee "$out/resources-after.txt"
helm history patch-lab --kube-context kiratech -n wordpress-build > "$out/helm-history.txt"

printf 'PASS: CPU change/restoration, image, upload hashes, volume identities and direct application content.\nEvidence: %s\n' "$out" | tee "$out/result.txt"
)
